  pkill -9 -f "python main.py"

Cancel All Orders:
  source venv/bin/activate
  python3 -c "
  import asyncio
  from lighter_client import get_client
  async def cancel_all():
      client = await get_client()
      await client.cancel_all_orders()
  asyncio.run(cancel_all())
  "
